    <script>
        const video = document.getElementById('webcam');
        navigator.mediaDevices.getUserMedia({ video: { width: 640, height: 480 } })
            .then(stream => { video.srcObject = stream; })
            .catch(err => alert("Không thể mở webcam: Vui lòng cấp quyền truy cập camera!"));

        const container = document.getElementById("threeContainer");
        const scene = new THREE.Scene();
        scene.background = new THREE.Color(0x090f1d);
        const camera3D = new THREE.PerspectiveCamera(45, container.clientWidth / container.clientHeight, 0.1, 1000);
        camera3D.position.set(50, 40, 50);

        const renderer = new THREE.WebGLRenderer({ antialias: true });
        renderer.setSize(container.clientWidth, container.clientHeight);
        container.appendChild(renderer.domElement);

        const controls = new THREE.OrbitControls(camera3D, renderer.domElement);
        controls.target.set(15, 12, 15);
        controls.update();

        const boxGeo = new THREE.BoxGeometry(30, 30, 30);
        const boxLines = new THREE.LineSegments(new THREE.EdgesGeometry(boxGeo), new THREE.LineBasicMaterial({ color: 0x1e3a5f }));
        boxLines.position.set(15, 15, 15);
        scene.add(boxLines);
        scene.add(new THREE.GridHelper(30, 10, 0x0284c7, 0x1e293b));

        const sphere = new THREE.Mesh(new THREE.SphereGeometry(1.5), new THREE.MeshBasicMaterial({ color: 0xef4444 }));
        sphere.position.set(15, 0, 15);
        scene.add(sphere);

        function animate() {
            requestAnimationFrame(animate);
            controls.update();
            renderer.render(scene, camera3D);
        }
        animate();

        let isCalibrating = false;
        let points = [];
        const overlay = document.getElementById("overlayCanvas");
        const ctx = overlay.getContext("2d");

        document.getElementById("btnCalib").onclick = () => {
            isCalibrating = true;
            points = [];
            overlay.width = video.videoWidth || 640;
            overlay.height = video.videoHeight || 480;
            ctx.clearRect(0, 0, overlay.width, overlay.height);
            alert("Click lần lượt 5 điểm trên khung hình:\n1. Gốc O(0,0,0)\n2. Điểm X(30,0,0)\n3. Điểm Chéo(30,30,0)\n4. Điểm Y(0,30,0)\n5. Đỉnh Z(0,0,30)");
        };

        overlay.onclick = (e) => {
            if (!isCalibrating) return;
            const rect = overlay.getBoundingClientRect();
            const x = (e.clientX - rect.left) * (overlay.width / rect.width);
            const y = (e.clientY - rect.top) * (overlay.height / rect.height);
            
            points.push({x, y});
            ctx.fillStyle = "#ef4444";
            ctx.beginPath(); ctx.arc(x, y, 5, 0, Math.PI * 2); ctx.fill();
            ctx.fillStyle = "#38bdf8"; ctx.font = "14px sans-serif";
            ctx.fillText("P" + points.length, x + 8, y - 8);

            if (points.length === 5) {
                isCalibrating = false;
                alert("Đã ghi nhận 5 điểm hiệu chuẩn thành công!");
            }
        };
    </script>
